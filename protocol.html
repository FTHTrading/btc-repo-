Reward Formula Specification

Eligible Reward = min(Base_Hours × Rate × Severity × Evidence, Event_Cap, Daily_Cap)
  • Base_Hours: Clamped strictly between 0.10h (6 minutes) and 24.00h (1,440 minutes).
  • Rate: Baseline rate coefficient set to 15.00 VTIME / hour.
  • Verification Multipliers: Self-Reported (1.00×), Device/Receipt Supported (1.25×), 3rd-Party Verified (1.50×), On-Chain Attested (1.75×).
  • Event Cap: Hard ceiling of 100.00 VTIME per individual action.
  • Daily Cap: Rolling 24-hour ceiling of 300.00 VTIME per participant.

EIP-712 Verifiable Claim Struct

struct RewardClaim {
    address recipient;
    uint256 amount;
    string receiptId;
    uint256 nonce;
    uint256 deadline;
    bytes signature;
}

function claimReward(RewardClaim calldata claim) external {
    require(block.timestamp <= claim.deadline, "Expired claim");
    require(!usedReceipt[claim.receiptId], "Receipt already claimed");
    bytes32 digest = _hashTypedDataV4(keccak256(abi.encode(
        REWARD_CLAIM_TYPEHASH,
        claim.recipient,
        claim.amount,
        keccak256(bytes(claim.receiptId)),
        claim.nonce,
        claim.deadline
    )));
    address signer = ECDSA.recover(digest, claim.signature);
    require(signer == trustedAttester, "Invalid signature");
    usedReceipt[claim.receiptId] = true;
    _mint(claim.recipient, claim.amount);
}
